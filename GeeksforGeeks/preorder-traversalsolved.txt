'''
# Node Class:
class Node:
    def __init__(self,val):
        self.data = val
        self.left = None
        self.right = None
'''

class Solution:
    def preOrder(self, root):
        ans = []

        def dfs(node):
            if node is None:
                return

            ans.append(node.data)
            dfs(node.left)
            dfs(node.right)

        dfs(root)
        return ans
