import heapq
class Solution:
    def minCost(self, arr):
        heapq.heapify(arr)
        cost=0
        while len(arr)>1:
            first=heapq.heappop(arr)
            second=heapq.heappop(arr)
            total=first+second
            cost+=total
            heapq.heappush(arr,total)
        return cost
