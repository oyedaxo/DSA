
class Solution(object):
    def subsets(self, nums):
        res = []
        
        def backtrack(index, current):
            res.append(list(current))
            for i in range(index, len(nums)):
                current.append(nums[i])
                backtrack(i + 1, current)
                current.pop()
                
        backtrack(0, [])
        return res
