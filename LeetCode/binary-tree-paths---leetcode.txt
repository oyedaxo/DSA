
    def binaryTreePaths(self, root):
        if not root:
            return []
            
        def dfs(node, path, result):
            if not node.left and not node.right:
                result.append(path + str(node.val))
                return
            if node.left:
                dfs(node.left, path + str(node.val) + "->", result)
            if node.right:
                dfs(node.right, path + str(node.val) + "->", result)
                
        paths = []
        dfs(root, "", paths)
        return paths
